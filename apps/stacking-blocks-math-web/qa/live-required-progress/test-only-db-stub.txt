/** 그 학급이 이 교사 것인지 확인한다. */
export async function teacherOwnsClass(
  db: any,
  teacherId: string,
  classId: string,
): Promise<boolean> {
  const { data } = await db
    .from("sb_classes")
    .select("id")
    .eq("id", classId)
    .eq("teacher_id", teacherId)
    .maybeSingle();
  return Boolean(data);
}
